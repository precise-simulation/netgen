cmake_minimum_required(VERSION 3.16)
project(NetgenNativeSdkSmoke LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(NETGEN_SDK_DIR "" CACHE PATH "Root of the extracted Netgen native SDK")
set(OCCT_INCLUDE_DIR "" CACHE PATH "Include directory of the pinned OCCT SDK")

if(NOT EXISTS "${NETGEN_SDK_DIR}/cmake/NetgenConfig.cmake")
  message(FATAL_ERROR "NETGEN_SDK_DIR does not contain cmake/NetgenConfig.cmake")
endif()
if(NOT EXISTS "${OCCT_INCLUDE_DIR}/Standard.hxx")
  message(FATAL_ERROR "OCCT_INCLUDE_DIR does not contain Standard.hxx")
endif()

find_package(Netgen CONFIG REQUIRED
  PATHS "${NETGEN_SDK_DIR}/cmake"
  NO_DEFAULT_PATH)

get_filename_component(_netgen_sdk_root "${NETGEN_SDK_DIR}" REALPATH)
file(TO_CMAKE_PATH "${_netgen_sdk_root}/" _netgen_sdk_prefix)
foreach(_target ngcore nglib)
  get_target_property(_configs ${_target} IMPORTED_CONFIGURATIONS)
  if(NOT _configs)
    set(_configs NOCONFIG)
  endif()
  set(_found_location FALSE)
  foreach(_config IN LISTS _configs)
    string(TOUPPER "${_config}" _config_upper)
    if(_config_upper STREQUAL "NOCONFIG")
      get_target_property(_artifact ${_target} IMPORTED_LOCATION)
    else()
      get_target_property(_artifact ${_target} "IMPORTED_LOCATION_${_config_upper}")
    endif()
    if(NOT _artifact)
      continue()
    endif()
    if(NOT EXISTS "${_artifact}")
      message(FATAL_ERROR "${_target} imported location does not exist: ${_artifact}")
    endif()
    get_filename_component(_artifact_real "${_artifact}" REALPATH)
    file(TO_CMAKE_PATH "${_artifact_real}" _artifact_path)
    string(FIND "${_artifact_path}" "${_netgen_sdk_prefix}" _artifact_prefix)
    if(NOT _artifact_prefix EQUAL 0)
      message(FATAL_ERROR "${_target} imported location resolves outside NETGEN_SDK_DIR: ${_artifact}")
    endif()
    set(_found_location TRUE)
  endforeach()
  if(NOT _found_location)
    message(FATAL_ERROR "${_target} has no usable imported location")
  endif()
endforeach()

add_library(netgen_native_sdk_headers OBJECT headers.cpp)
target_compile_definitions(netgen_native_sdk_headers PRIVATE OCCGEOMETRY)
target_include_directories(netgen_native_sdk_headers PRIVATE "${OCCT_INCLUDE_DIR}")
target_link_libraries(netgen_native_sdk_headers PRIVATE nglib)

add_executable(netgen_native_sdk_smoke main.cpp)
target_link_libraries(netgen_native_sdk_smoke PRIVATE nglib)
